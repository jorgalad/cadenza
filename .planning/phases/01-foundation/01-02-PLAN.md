         __init__(self, source: str): store source, position=0, line=1, column=1, tokens=[], warnings=[]
         Method tokenize() -> tuple[list[Token], list[ParseWarning]]:
           Main loop: skip whitespace, then match in priority order:
           a. '(' -> LPAREN
           b. ')' -> RPAREN
           c. '-' followed by optional digit (tuplet) + duration letter + optional dots -> REST
           d. Digit 2-9 followed by duration letter + optional dots -> DURATION (tuplet)
           e. Duration letter NOT followed by digit (to disambiguate from pitch):

              - 's' not followed by a-g letter (to disambiguate from 'stacc') -> DURATION sixteenth
              - 'w', 'h', 'q', 't', 'x' not followed by digit -> DURATION
              - Append dots if present
           f. Pitch letter followed by optional accidental ('s','b','ss','bb','n') + digit(s) -> PITCH
              - For chords: if immediately followed (no space) by another pitch pattern, continue accumulating into same token value
              - Critical: 'e' followed by optional accidental + digit = pitch E, 'f' followed by optional accidental + digit = pitch F
           g. Alphabetic string matching DYNAMICS (longest match first: "fffff" before "ffff" before "fff" etc.) -> DYNAMIC
           h. Alphabetic string matching KNOWN_ARTICULATIONS -> ARTICULATION
           i. Any other alphabetic string -> ARTICULATION + emit ParseWarning("Unknown articulation: '{value}'")
           j. Anything else -> raise ParseError with position info

         Helper _skip_whitespace(): advance past spaces, tabs, newlines; update line/column
         Helper _peek() -> str | None
         Helper _peek_ahead(n) -> str | None
         Helper _advance() -> str
         Helper _read_while(predicate) -> str

    Key disambiguation logic:
    - At each position, try to match the longest valid token
    - For 'e': look ahead -- if next char is a digit (0-9), this starts a pitch (e4, e5). If next char is a dot or whitespace or ')' or another letter, this is duration eighth.
    - For 'f': look ahead -- if next char is a digit or accidental-starting char followed by digit, this is a pitch. Otherwise check if context is dynamic position (after a pitch token, before next pitch/duration).

    1. Create src/cadenza/omn/parser.py:
       - class OmnParser:
         __init__(self, source: str):
           self._source = source
           self._tokenizer = OmnTokenizer(source)
           self._tokens, self._tokenizer_warnings = self._tokenizer.tokenize()
           self._pos = 0
           self._warnings: list[ParseWarning] = list(self._tokenizer_warnings)
           # Sticky state
           self._current_duration: Duration | None = None
           self._current_dynamic: str | None = None
           self._current_articulations: tuple[str, ...] = ()

         Method parse() -> tuple[Phrase, list[ParseWarning]]:
           events: list[Event] = []
           while not self._at_end():
             if self._check(TokenType.LPAREN):
               events.extend(self._parse_group())
             else:
               event = self._parse_event()
               if event is not None:
                 events.append(event)
           return tuple(events), self._warnings

         Method _parse_event() -> Event:
           # 1. Check for duration token -> update sticky duration
           if self._check(TokenType.DURATION):
             token = self._advance()
             self._current_duration = self._parse_duration_value(token.value)
           # 2. Check for rest
           if self._check(TokenType.REST):
             token = self._advance()
             rest_dur = self._parse_rest_value(token.value)
             return Rest(duration=rest_dur)
           # 3. Expect pitch
           if not self._check(TokenType.PITCH):
             token = self._peek_token()
             raise ParseError(message=f"Expected pitch, got {token.type.name}: '{token.value}'", position=token.position, line=token.line, column=token.column, source_snippet=self._snippet(token.position))
           pitch_token = self._advance()
           pitch = self._parse_pitch_value(pitch_token.value)
           # 4. Consume optional dynamic
           if self._check(TokenType.DYNAMIC):
             self._current_dynamic = self._advance().value
           # 5. Consume optional articulations (one or more)
           new_arts: list[str] = []
           while self._check(TokenType.ARTICULATION):
             new_arts.append(self._advance().value)
           if new_arts:
             self._current_articulations = tuple(new_arts)
           # 6. Build Note with sticky values
           duration = self._current_duration if self._current_duration is not None else Duration.from_omn("q")
           return Note(pitch=pitch, duration=duration, dynamic=self._current_dynamic, articulations=self._current_articulations)

         Method _parse_group() -> list[Event]:
           self._expect(TokenType.LPAREN)
           events = []
           while not self._check(TokenType.RPAREN) and not self._at_end():
             event = self._parse_event()
             if event is not None:
               events.append(event)
           self._expect(TokenType.RPAREN)
           return events

         Method _parse_duration_value(value: str) -> Duration:
           Parse tuplet prefix (digit at start), base letter, dots (trailing periods)
           Return Duration.from_omn(base, dots=dot_count, tuplet=tuplet_value)

         Method _parse_rest_value(value: str) -> Duration:
           Strip leading '-', then parse as duration value

         Method _parse_pitch_value(value: str) -> Pitch:
           Extract letter (first char), accidental (s/b/ss/bb/n or default 'n'), octave (remaining digits)
           Return Pitch(step=letter, accidental=acc, octave=int(octave_str))
           Note: for chord tokens (c4e4g4) -- for Phase 1, parse only the first pitch and emit warning about chords being deferred. OR parse all pitches but return only first with warning.

         Helpers: _at_end, _check(type), _advance, _expect(type), _peek_token, _snippet(pos)

       - Convenience function: def parse_omn(source: str) -> tuple[Phrase, list[ParseWarning]]:
           return OmnParser(source).parse()

       - def to_omn(phrase: Phrase) -> str:
           Track prev_duration, prev_dynamic, prev_articulations (all start as None/empty)
           For each event:
             If Rest: emit "-" + _duration_to_omn(event.duration). Reset is NOT sticky (previous note state carries through).
             If Note:
               tokens = []
               If event.duration != prev_duration: tokens.append(_duration_to_omn(event.duration)); prev_duration = event.duration
               tokens.append(_pitch_to_omn(event.pitch))  # always emit pitch
               If event.dynamic != prev_dynamic: tokens.append(event.dynamic) if not None; prev_dynamic = event.dynamic
               If event.articulations != prev_articulations: tokens.extend(event.articulations); prev_articulations = event.articulations
               Append " ".join(tokens) to parts
           Return " ".join(parts)

       - def _duration_to_omn(d: Duration) -> str:
           result = ""
           if d.tuplet is not None: result += str(d.tuplet)
           result += d.base
           result += "." * d.dots
           return result

           acc = "" if p.accidental == "n" else p.accidental
